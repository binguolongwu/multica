# Chat V2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the floating chat FAB/window with a first-class IM-style Chat tab (dual-pane: thread list + conversation), with read cursors, unread badges, session/agent pinning, welcome chats, and retained floating window mode toggle.

**Architecture:** Vertical slice implementation across 6 slices. Each slice delivers a complete, testable feature from DB → API → Core → Views → Routing. Shared `packages/views/chat/` components serve both Web (`app/.../chat/page.tsx`) and Desktop (`routes.tsx`). Existing `chat-input.tsx`, `chat-message-list.tsx` are reused directly in the new dual-pane layout.

**Tech Stack:** TypeScript strict, React, Zustand, TanStack Query, Go/Chi, sqlc, PostgreSQL

## Global Constraints

- Web + Desktop must have identical Chat V2 experience
- Existing floating chat must remain functional and not be broken
- New Chat nav item must be a single word: `chat` (per CLAUDE.md route naming rules)
- All API JSON responses must pass through `parseWithFallback` + zod schema (per CLAUDE.md API compatibility rules)
- Mutations must be optimistic: patch locally, roll back on failure, invalidate on settle
- Workspace-scoped query keys must include `wsId`
- Zustand `chatMode` must persist via `StorageAdapter`, not directly to localStorage
- `packages/views/` must not import `next/*` or `react-router-dom`; use `useNavigation()`, `<AppLink>`
- `@multica/core` types only; no UI imports in core
- New tables must use business-meaningful PK names
- Desktop pre-workspace flows register `WindowOverlay` types; chat is a session route, not a transition flow

```go
func deriveChatTitle(content string, agentName string) string {
    // Strip markdown
    re := regexp.MustCompile(`[#*>` + "`" + `]|!\[.*?\]\(.*?\)|\[.*?\]\(.*?\)`)
    cleaned := re.ReplaceAllString(content, "")
    cleaned = strings.TrimSpace(cleaned)
    // Collapse whitespace
    spaceRE := regexp.MustCompile(`\s+`)
    cleaned = spaceRE.ReplaceAllString(cleaned, " ")
    // Truncate to 30 chars
    runes := []rune(cleaned)
    if len(runes) > 30 {
        cleaned = string(runes[:30])
    }
    if cleaned == "" {
        return "Chat with " + agentName
    }
    return cleaned
}
```

```go
package handler

import (

func (h *Handler) ListPinnedAgents(w http.ResponseWriter, r *http.Request) {
    workspaceID := h.resolveWorkspaceID(r)
    member, ok := h.workspaceMember(w, r, workspaceID)
    if !ok { return }

    agents, err := h.Queries.ListPinnedAgents(r.Context(), db.ListPinnedAgentsParams{
        UserID:      member.UserID,
        WorkspaceID: parseUUID(workspaceID),
    })
    if err != nil {
        slog.Warn("chat: failed to list pinned agents", "error", err)
        writeError(w, http.StatusInternalServerError, "failed to list pinned agents")
        return
    }
    writeJSON(w, http.StatusOK, agents)
}

func (h *Handler) PinAgent(w http.ResponseWriter, r *http.Request) {
    workspaceID := h.resolveWorkspaceID(r)
    member, ok := h.workspaceMember(w, r, workspaceID)
    if !ok { return }

    var req struct {
        AgentID    string `json:"agent_id"`
        SortOrder  int16  `json:"sort_order"`
    }
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        writeError(w, http.StatusBadRequest, "invalid request body")
        return
    }

    agentID, ok := parseUUIDOrBadRequest(w, req.AgentID, "agent_id")
    if !ok { return }

    // Enforce max 5 pinned agents
    count, err := h.Queries.CountPinnedAgents(r.Context(), db.CountPinnedAgentsParams{
        UserID:      member.UserID,
        WorkspaceID: parseUUID(workspaceID),
    })
    if err != nil {
        writeError(w, http.StatusInternalServerError, "failed to count pinned agents")
        return
    }
    if count >= 5 {
        writeError(w, http.StatusBadRequest, "maximum 5 pinned agents")
        return
    }

    err = h.Queries.PinAgent(r.Context(), db.PinAgentParams{
        UserID:      member.UserID,
        AgentID:     agentID,
        WorkspaceID: parseUUID(workspaceID),
        SortOrder:   req.SortOrder,
    })
    if err != nil {
        slog.Warn("chat: failed to pin agent", "error", err)
        writeError(w, http.StatusInternalServerError, "failed to pin agent")
        return
    }
    w.WriteHeader(http.StatusCreated)
    writeJSON(w, http.StatusCreated, map[string]string{"ok": "true"})
}

func (h *Handler) UnpinAgent(w http.ResponseWriter, r *http.Request) {
    workspaceID := h.resolveWorkspaceID(r)
    member, ok := h.workspaceMember(w, r, workspaceID)
    if !ok { return }

    agentID, ok := parseUUIDOrBadRequest(w, chi.URLParam(r, "agentId"), "agent_id")
    if !ok { return }

    err := h.Queries.UnpinAgent(r.Context(), db.UnpinAgentParams{
        UserID:      member.UserID,
        AgentID:     agentID,
        WorkspaceID: parseUUID(workspaceID),
    })
    if err != nil {
        slog.Warn("chat: failed to unpin agent", "error", err)
        writeError(w, http.StatusInternalServerError, "failed to unpin agent")
        return
    }
    w.WriteHeader(http.StatusNoContent)
}
```

```typescript
export function useToggleSessionPin(wsId: string) {
  const queryClient = useQueryClient();
  return useMutation({
    mutationFn: (sessionId: string) => api.toggleSessionPin(wsId, sessionId),
    onMutate: async (sessionId) => {
      await queryClient.cancelQueries({ queryKey: chatKeys.sessions(wsId) });
      const prev = queryClient.getQueryData<ChatSession[]>(chatKeys.sessions(wsId));
      if (prev) {
        queryClient.setQueryData<ChatSession[]>(chatKeys.sessions(wsId),
          prev.map(s => s.id === sessionId ? { ...s, is_pinned: !s.is_pinned } : s)
        );
      }
      return { prev };
    },
    onError: (_err, _vars, ctx) => {
      if (ctx?.prev) {
        queryClient.setQueryData(chatKeys.sessions(wsId), ctx.prev);
      }
    },
    onSettle: () => {
      queryClient.invalidateQueries({ queryKey: chatKeys.sessions(wsId) });
    },
  });
}
```

```tsx
              if (e.key === "Enter") handleSubmit();
              if (e.key === "Escape") { setDraft(session.title); setEditing(false); }
            }}
            onBlur={handleSubmit}
            className="w-full text-sm font-medium bg-transparent border-b border-primary outline-none"
```

```tsx
"use client";

import React from "react";
import { useChatStore } from "@multica/core/chat";
import { ChatFab } from "./components/chat-fab";
import { ChatWindow } from "./components/chat-window";

/**
 * Mode orchestrator: renders floating chat (FAB + Window) only when
 * chatMode is 'floating'. In 'tab' mode, the ChatPage handles display.
 */
export function FloatingChat() {
  const chatMode = useChatStore((s) => s.chatMode);

  if (chatMode !== "floating") return null;

  return (
    <>
      <ChatFab />
      <ChatWindow />
    </>
  );
}
```
